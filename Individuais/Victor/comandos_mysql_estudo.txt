1. DDL (DATA DEFINITION LANGUAGE) - CRIAÇÃO DE TABELAS DO PROJETO
--------------------------------------------------------------------------------

CREATE TABLE IF NOT EXISTS empresa (
    id INT PRIMARY KEY AUTO_INCREMENT,
    razao_social VARCHAR(100) NOT NULL,
    cnpj CHAR(14) UNIQUE NOT NULL,
    codigo_ativacao CHAR(8) UNIQUE NOT NULL,
    endereco VARCHAR(150),
    id_matriz INT,
    CONSTRAINT fk_empresa_matriz FOREIGN KEY (id_matriz) REFERENCES empresa(id)
);

CREATE TABLE IF NOT EXISTS usuario (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(70) NOT NULL,
    email VARCHAR(70) UNIQUE NOT NULL,
    senha VARCHAR(45) NOT NULL,
    id_empresa INT NOT NULL,
    CONSTRAINT fk_empresa_usuario FOREIGN KEY (id_empresa) REFERENCES empresa(id)
);

CREATE TABLE IF NOT EXISTS camara (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(45) NOT NULL,
    id_empresa INT NOT NULL,
    CONSTRAINT fk_empresa_camara FOREIGN KEY (id_empresa) REFERENCES empresa(id)
);

CREATE TABLE IF NOT EXISTS sensor (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nome_sensor VARCHAR(45) NOT NULL,
    status_ VARCHAR(20) DEFAULT 'ativo',
    id_camara INT NOT NULL,
    CONSTRAINT chk_status CHECK (status_ IN ('ativo', 'inativo', 'manutencao')),
    CONSTRAINT fk_camara_sensor FOREIGN KEY (id_camara) REFERENCES camara(id)
);

CREATE TABLE IF NOT EXISTS leitura (
    id INT PRIMARY KEY AUTO_INCREMENT,
    temperatura FLOAT NOT NULL,
    momento DATETIME DEFAULT CURRENT_TIMESTAMP,
    id_sensor INT NOT NULL,
    CONSTRAINT fk_sensor_leitura FOREIGN KEY (id_sensor) REFERENCES sensor(id)
);

4. DQL - SELECTS BÁSICOS E FILTROS (WHERE, OPERADORES, STRINGS)
--------------------------------------------------------------------------------

-- 4.1 Selecionar todas as colunas de uma tabela:
SELECT * FROM empresa;
SELECT * FROM usuario;
SELECT * FROM camara;
SELECT * FROM sensor;
SELECT * FROM leitura;

-- 4.2 Selecionar colunas específicas com Alias (apelidos):
SELECT nome_sensor AS 'Sensor', status_ AS 'Situação Atual' FROM sensor;
SELECT razao_social AS 'Nome da Empresa', cnpj AS 'Documento CNPJ' FROM empresa;

-- 4.3 Filtro simples com WHERE:
SELECT * FROM sensor WHERE status_ = 'ativo';
SELECT * FROM sensor WHERE status_ = 'manutencao';
SELECT * FROM leitura WHERE temperatura > 8.0;

-- 4.4 Filtro composto com AND, OR, NOT:
SELECT * FROM leitura WHERE temperatura >= 2.0 AND temperatura <= 8.0;
SELECT * FROM sensor WHERE status_ = 'manutencao' OR status_ = 'inativo';
SELECT * FROM sensor WHERE NOT status_ = 'ativo';

-- 4.5 Filtro com BETWEEN (faixa de valores):
-- Leituras na faixa ideal de conservação de vacinas (2°C a 8°C):
SELECT * FROM leitura WHERE temperatura BETWEEN 2.0 AND 8.0;

-- Leituras fora da faixa ideal (alertas):
SELECT * FROM leitura WHERE temperatura NOT BETWEEN 2.0 AND 8.0;

-- 4.6 Filtro com IN (lista de opções aceitas):
SELECT * FROM sensor WHERE status_ IN ('manutencao', 'inativo');
SELECT * FROM camara WHERE id_empresa IN (1, 2);

-- 4.7 Filtro de strings com LIKE (padrões de busca):
-- Começa com determinado texto (ex: 'Farmacêutica%'):
SELECT * FROM empresa WHERE razao_social LIKE 'Farmacêutica%';

-- Termina com determinado texto (ex: '%Ltda'):
SELECT * FROM empresa WHERE razao_social LIKE '%Ltda';

-- Contém a palavra em qualquer posição:
SELECT * FROM usuario WHERE email LIKE '%@farmavacina.com%';

-- Caractere curinga sublinhado '_' (representa exatamente 1 caractere):
SELECT * FROM sensor WHERE nome_sensor LIKE 'LM35-__';

-- 4.8 Tratamento de valores Nulos (IS NULL e IS NOT NULL):
-- Empresas que são Matrizes (não têm id_matriz):
SELECT * FROM empresa WHERE id_matriz IS NULL;

-- Empresas que são Filiais (possuem id_matriz):
SELECT * FROM empresa WHERE id_matriz IS NOT NULL;

-- 4.9 Ordenação de resultados (ORDER BY ASC / DESC):
-- Ordenar por temperatura da maior para a menor:
SELECT * FROM leitura ORDER BY temperatura DESC;

-- Ordenar por momento mais recente primeiro:
SELECT * FROM leitura ORDER BY momento DESC;

-- Ordenar usuários por nome em ordem alfabética:
SELECT * FROM usuario ORDER BY nome ASC;

-- 4.10 Limitar quantidade de resultados (LIMIT e OFFSET):
-- Trazer apenas os 3 primeiros registros:
SELECT * FROM leitura LIMIT 3;

-- Trazer o último registro de leitura feito:
SELECT * FROM leitura ORDER BY id DESC LIMIT 1;

-- Paginação: pular os 2 primeiros e trazer os próximos 3:
SELECT * FROM leitura LIMIT 3 OFFSET 2;

-- 4.11 Remover duplicatas com DISTINCT:
SELECT DISTINCT status_ FROM sensor;
SELECT DISTINCT id_empresa FROM camara;

5. FUNÇÕES DE AGREGAÇÃO E AGRUPAMENTO (COUNT, SUM, AVG, MIN, MAX, GROUP BY)
--------------------------------------------------------------------------------

-- Quantidade total de registros:
SELECT COUNT(*) AS total_leituras FROM leitura;
SELECT COUNT(*) AS total_sensores_ativos FROM sensor WHERE status_ = 'ativo';

-- Maior e Menor temperatura registrada:
SELECT MAX(temperatura) AS temp_maxima, MIN(temperatura) AS temp_minima FROM leitura;

-- Média de temperatura com arredondamento (ROUND):
SELECT ROUND(AVG(temperatura), 2) AS media_temperatura FROM leitura;

-- Soma de valores (exemplo geral):
SELECT SUM(temperatura) AS soma_temperaturas FROM leitura;

-- 5.1 Agrupamento com GROUP BY:
-- Média, Máxima e Mínima de temperatura por Sensor:
SELECT 
    id_sensor,
    COUNT(*) AS total_registros,
    ROUND(AVG(temperatura), 2) AS media_temp,
    MIN(temperatura) AS min_temp,
    MAX(temperatura) AS max_temp
FROM leitura
GROUP BY id_sensor;

-- Quantidade de sensores por Status:
SELECT status_, COUNT(*) AS quantidade FROM sensor GROUP BY status_;

-- Quantidade de câmaras por Empresa:
SELECT id_empresa, COUNT(*) AS total_camaras FROM camara GROUP BY id_empresa;

-- 5.2 Filtro após agrupamento com HAVING:
-- Exibir apenas sensores que tenham mais de 2 leituras registradas:
SELECT id_sensor, COUNT(*) AS total_leituras
FROM leitura
GROUP BY id_sensor
HAVING COUNT(*) > 2;

-- Sensores cuja média de temperatura ultrapassou 5°C:
SELECT id_sensor, ROUND(AVG(temperatura), 2) AS media_temp
FROM leitura
GROUP BY id_sensor
HAVING AVG(temperatura) > 5.0;

6. JOINS - RELACIONAMENTO ENTRE TABELAS (O QUE O PROFESSOR MAIS COBRA)
--------------------------------------------------------------------------------

-- 6.1 INNER JOIN: Apenas registros que possuem correspondência em ambas as tabelas
-- Usuário e a Razão Social da Empresa dele:
SELECT 
    u.id AS id_usuario,
    u.nome AS nome_usuario,
    u.email,
    e.razao_social AS empresa
FROM usuario AS u
INNER JOIN empresa AS e ON u.id_empresa = e.id;

-- Câmaras e seus respectivos Sensores:
SELECT 
    c.nome AS camara,
    s.nome_sensor,
    s.status_
FROM camara AS c
INNER JOIN sensor AS s ON s.id_camara = c.id;

-- Leituras com nome do Sensor e data formatada:
SELECT 
    l.id AS id_leitura,
    s.nome_sensor,
    l.temperatura,
    DATE_FORMAT(l.momento, '%d/%m/%Y %H:%i:%s') AS momento_formatado
FROM leitura AS l
INNER JOIN sensor AS s ON l.id_sensor = s.id;

-- 6.2 INNER JOIN MULTIPLO (Cruzando todas as tabelas: Empresa -> Câmara -> Sensor -> Leitura):
SELECT 
    e.razao_social AS empresa,
    c.nome AS camara,
    s.nome_sensor,
    l.temperatura,
    l.momento
FROM empresa AS e
INNER JOIN camara AS c ON c.id_empresa = e.id
INNER JOIN sensor AS s ON s.id_camara = c.id
INNER JOIN leitura AS l ON l.id_sensor = s.id
ORDER BY l.momento DESC;

-- 6.3 LEFT JOIN: Traz todos da tabela da esquerda, mesmo se não tiver correspondente na direita
-- Trazer todas as câmaras, inclusive aquelas que ainda não têm nenhum sensor cadastrado:
SELECT 
    c.id AS id_camara,
    c.nome AS camara,
    s.nome_sensor,
    s.status_
FROM camara AS c
LEFT JOIN sensor AS s ON s.id_camara = c.id;

-- Trazer todos os sensores e suas leituras (mesmo sensores que ainda não tiveram nenhuma leitura):
SELECT 
    s.nome_sensor,
    s.status_,
    l.temperatura,
    l.momento
FROM sensor AS s
LEFT JOIN leitura AS l ON l.id_sensor = s.id;

-- Descobrir câmaras sem nenhum sensor (técnica de auditoria / anti-join):
SELECT 
    c.nome AS camara_sem_sensor
FROM camara AS c
LEFT JOIN sensor AS s ON s.id_camara = c.id
WHERE s.id IS NULL;

-- 6.4 RIGHT JOIN: Traz todos da tabela da direita, mesmo se não tiver correspondente na esquerda
SELECT 
    s.nome_sensor,
    c.nome AS camara
FROM camara AS c
RIGHT JOIN sensor AS s ON s.id_camara = c.id;

-- 6.5 SELF JOIN (Auto-relacionamento): Empresa Filial e sua respectiva Matriz:
SELECT 
    filial.razao_social AS 'Nome Filial',
    filial.cnpj AS 'CNPJ Filial',
    matriz.razao_social AS 'Nome Matriz',
    matriz.cnpj AS 'CNPJ Matriz'
FROM empresa AS filial
INNER JOIN empresa AS matriz ON filial.id_matriz = matriz.id;

-- Trazer todas as empresas mostrando quem é a matriz (usando LEFT JOIN para a matriz aparecer com NULL):
SELECT 
    e.razao_social AS empresa,
    IFNULL(m.razao_social, 'MATRIZ PRINCIPAL') AS matriz_vinculada
FROM empresa AS e
LEFT JOIN empresa AS m ON e.id_matriz = m.id;

7. CONDICIONAIS E ESTATÍSTICAS AVANÇADAS (CASE WHEN, IFNULL, CONCAT)
--------------------------------------------------------------------------------

-- Classificação de temperatura de acordo com as normas da Anvisa para vacinas:
SELECT 
    l.id,
    s.nome_sensor,
    l.temperatura,
    l.momento,
    CASE 
        WHEN l.temperatura < 2.0 THEN 'ALERTA: Congelamento (Abaixo de 2°C)'
        WHEN l.temperatura > 8.0 THEN 'ALERTA: Superaquecimento (Acima de 8°C)'
        ELSE 'TEMPERATURA IDEAL (2°C a 8°C)'
    END AS status_temperatura
FROM leitura AS l
INNER JOIN sensor AS s ON l.id_sensor = s.id;

-- Concatenar textos com CONCAT:
SELECT 
    CONCAT(nome, ' - Contato: ', email) AS info_usuario 
FROM usuario;

SELECT 
    CONCAT(temperatura, ' °C') AS temp_com_unidade 
FROM leitura;

-- Tratamento de campos nulos com IFNULL ou COALESCE:
SELECT 
    razao_social, 
    IFNULL(id_matriz, 'Sem Matriz') AS status_matriz 
FROM empresa;

8. SUBQUERIES (SUB-CONSULTAS)
--------------------------------------------------------------------------------

SELECT * 
FROM leitura

SELECT * 
FROM sensor 
WHERE id_camara IN (
    SELECT id FROM camara WHERE id_empresa = 1
);

SELECT DISTINCT nome_sensor

WHERE id IN (
    SELECT id_sensor FROM leitura WHERE temperatura > 8.0

--------------------------------------------------------------------------------
11. LIMPEZA TOTAL DE TABELAS (TRUNCATE TABLE vs DELETE)
--------------------------------------------------------------------------------

-- DIFERENÇA EXPLICADA PARA O PROFESSOR:
-- DELETE FROM tabela; -> Apaga linha por linha, gera log, mantém o AUTO_INCREMENT de onde parou.
-- TRUNCATE TABLE tabela; -> Recria a tabela vazia, é DDL, muito mais rápido, reseta o AUTO_INCREMENT para 1.

-- Para truncar tabelas com Foreign Keys sem erro de constraint no MySQL:
SET FOREIGN_KEY_CHECKS = 0;

-- Resetar a tabela de leituras zerando o AUTO_INCREMENT:
TRUNCATE TABLE leitura;

-- Resetar a tabela de sensores zerando o AUTO_INCREMENT:
TRUNCATE TABLE sensor;

-- Reativar a verificação de chaves estrangeiras:
SET FOREIGN_KEY_CHECKS = 1;

12. VIEWS (VISÕES) - TABELAS VIRTUAIS EXCELENTES PARA DASHBOARDS
--------------------------------------------------------------------------------

-- Criar uma View que junta todo o monitoramento para o front-end:
CREATE OR REPLACE VIEW vw_monitoramento_geral AS
SELECT 
    e.razao_social AS empresa,
    c.nome AS camara,
    s.nome_sensor AS sensor,
    s.status_ AS status_sensor,
    l.temperatura,
    l.momento,
    CASE 
        WHEN l.temperatura < 2.0 THEN 'CRÍTICO BAIXO'
        WHEN l.temperatura > 8.0 THEN 'CRÍTICO ALTO'
        ELSE 'NORMAL'
    END AS situacao
FROM empresa AS e
INNER JOIN camara AS c ON c.id_empresa = e.id
INNER JOIN sensor AS s ON s.id_camara = c.id
INNER JOIN leitura AS l ON l.id_sensor = s.id;

-- Como consultar a View (funciona exatamente como se fosse uma tabela!):
SELECT * FROM vw_monitoramento_geral;

-- Filtrar alertas usando a View:
SELECT * FROM vw_monitoramento_geral WHERE situacao <> 'NORMAL';
